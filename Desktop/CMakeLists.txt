cmake_minimum_required(VERSION 3.22)
project(LightTouchDesktopUI LANGUAGES CXX)
find_package(SDL2 REQUIRED)
add_library(lighttouch_desktop_ui INTERFACE)
target_compile_features(lighttouch_desktop_ui INTERFACE cxx_std_17)
target_include_directories(lighttouch_desktop_ui INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/ui")
target_link_libraries(lighttouch_desktop_ui INTERFACE SDL2::SDL2)
add_library(lighttouch_desktop_window STATIC native/Window.cpp)
target_include_directories(lighttouch_desktop_window PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native/include")
target_link_libraries(lighttouch_desktop_window PRIVATE lighttouch_desktop_ui)
option(LIGHTTOUCH_DESKTOP_UI_TESTS "Asset-free shared SDL UI contract tests" OFF)
if(LIGHTTOUCH_DESKTOP_UI_TESTS)
 enable_testing()
 add_executable(lighttouch_desktop_controls_test tests/ControlBar.cpp)
 target_link_libraries(lighttouch_desktop_controls_test PRIVATE lighttouch_desktop_ui)
 if(MSVC)
  target_compile_options(lighttouch_desktop_controls_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(lighttouch_desktop_controls_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME lighttouch_desktop_controls COMMAND lighttouch_desktop_controls_test)
 add_executable(lighttouch_desktop_native_window_test tests/NativeWindow.cpp)
 target_link_libraries(lighttouch_desktop_native_window_test PRIVATE lighttouch_desktop_ui)
 if(MSVC)
  target_compile_options(lighttouch_desktop_native_window_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(lighttouch_desktop_native_window_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME lighttouch_desktop_native_window COMMAND lighttouch_desktop_native_window_test)
endif()
