cmake_minimum_required(VERSION 3.22)
project(LightTouchLinux LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_subdirectory(../Portable portable)
find_package(SDL2 REQUIRED)
add_executable(lighttouch-linux Main.cpp)
target_link_libraries(lighttouch-linux PRIVATE lighttouch_runtime SDL2::SDL2)
target_compile_options(lighttouch-linux PRIVATE -Wall -Wextra -Werror)
if(LIGHTTOUCH_BUILD_TESTS)
 add_executable(lighttouch_presentation_test tests/Presentation.cpp)
 target_link_libraries(lighttouch_presentation_test PRIVATE SDL2::SDL2)
 target_compile_options(lighttouch_presentation_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_presentation COMMAND lighttouch_presentation_test)
 add_executable(lighttouch_admission_test tests/Admission.cpp)
 target_link_libraries(lighttouch_admission_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_admission_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_admission COMMAND lighttouch_admission_test)
 add_executable(lighttouch_script_stop_test tests/ScriptStop.cpp)
 target_compile_options(lighttouch_script_stop_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_script_stop COMMAND lighttouch_script_stop_test)
 add_executable(lighttouch_adapter_lifecycle_test tests/AdapterLifecycle.cpp)
 target_link_libraries(lighttouch_adapter_lifecycle_test PRIVATE lighttouch_runtime)
 target_compile_options(lighttouch_adapter_lifecycle_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME lighttouch_adapter_lifecycle COMMAND lighttouch_adapter_lifecycle_test)
endif()
