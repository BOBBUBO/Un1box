cmake_minimum_required(VERSION 3.18)
project(LightTouchFirmware LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT XPWN_SOURCE)
  message(FATAL_ERROR "XPWN_SOURCE must name the pinned XPwn source (see README.md)")
endif()
execute_process(COMMAND git -C "${XPWN_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE XPWN_REV OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE XPWN_GIT_RESULT)
if(NOT XPWN_GIT_RESULT EQUAL 0 OR NOT XPWN_REV STREQUAL "20c32e5c12d1b22a9d55a59a0ff6267f539b77f4")
  message(FATAL_ERROR "XPwn source pin mismatch")
endif()
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED)
add_compile_definitions(HAVE_CRYPT _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE)
include_directories(${XPWN_SOURCE}/includes)
add_library(ltf_common STATIC ${XPWN_SOURCE}/common/abstractfile.c ${XPWN_SOURCE}/common/base64.c)
# XPwn removes a nonexistent key before its first attribute insertion; an
# empty tree has no root node to traverse. Keep this pinned fix in the build
# recipe and corresponding sources rather than modifying a shared checkout.
file(READ "${XPWN_SOURCE}/hfs/btree.c" HFS_BTREE_SOURCE)
string(REPLACE "int removeFromBTree(BTree* tree, BTKey* searchKey) {" "int removeFromBTree(BTree* tree, BTKey* searchKey) {\n  if(tree->headerRec->rootNode == 0) return TRUE;" HFS_BTREE_SOURCE "${HFS_BTREE_SOURCE}")
string(REPLACE "void* search(BTree* tree, BTKey* searchKey, int *exact, uint32_t *nodeNumber, int *recordNumber) {" "void* search(BTree* tree, BTKey* searchKey, int *exact, uint32_t *nodeNumber, int *recordNumber) {\n  if(tree->headerRec->rootNode == 0) { if(exact) *exact=FALSE; if(nodeNumber) *nodeNumber=0; if(recordNumber) *recordNumber=0; return NULL; }" HFS_BTREE_SOURCE "${HFS_BTREE_SOURCE}")
string(REPLACE "if(lastRecordDataOffset == 0) {\n    hfs_panic(\"BTree inconsistent!\");\n    return NULL;\n  }" "if(lastRecordDataOffset == 0) {\n    if(descriptor->numRecords == 0) { free(descriptor); if(exact) *exact=FALSE; if(nodeNumber) *nodeNumber=root; if(recordNumber) *recordNumber=0; return NULL; }\n    lastRecordDataOffset=recordDataOffset;\n  }" HFS_BTREE_SOURCE "${HFS_BTREE_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hfs-btree.c" "${HFS_BTREE_SOURCE}")
list(APPEND HFS_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/hfs-btree.c")
file(READ "${XPWN_SOURCE}/hfs/rawfile.c" HFS_RAW_SOURCE)
string(FIND "${HFS_RAW_SOURCE}" "int writeExtents(RawFile* rawFile) {" HFS_WRITE_START)
string(FIND "${HFS_RAW_SOURCE}" "int readExtents(RawFile* rawFile) {" HFS_WRITE_END)
if(HFS_WRITE_START LESS 0 OR HFS_WRITE_END LESS HFS_WRITE_START)
  message(FATAL_ERROR "Pinned XPwn extent patch no longer matches")
endif()
string(SUBSTRING "${HFS_RAW_SOURCE}" 0 ${HFS_WRITE_START} HFS_RAW_PREFIX)
string(SUBSTRING "${HFS_RAW_SOURCE}" ${HFS_WRITE_END} -1 HFS_RAW_SUFFIX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/xpwn-write-extents.inc" HFS_WRITE_FIXED)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hfs-rawfile.c" "${HFS_RAW_PREFIX}${HFS_WRITE_FIXED}${HFS_RAW_SUFFIX}")
list(APPEND HFS_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/hfs-rawfile.c")
set(HFS_FILES catalog extents xattr fastunicodecompare flatfile hfslib utility volume hfscompress)
foreach(file IN LISTS HFS_FILES)
  list(APPEND HFS_SOURCES ${XPWN_SOURCE}/hfs/${file}.c)
endforeach()
add_library(ltf_hfs STATIC ${HFS_SOURCES})
target_link_libraries(ltf_hfs ltf_common ZLIB::ZLIB)
set(DMG_FILES checksum dmgfile dmglib filevault io partition resources udif)
foreach(file IN LISTS DMG_FILES)
  list(APPEND DMG_SOURCES ${XPWN_SOURCE}/dmg/${file}.c)
endforeach()
add_library(ltf_dmg STATIC ${DMG_SOURCES})
target_link_libraries(ltf_dmg ltf_common ltf_hfs ZLIB::ZLIB OpenSSL::Crypto)

add_library(ltf_zip STATIC ${XPWN_SOURCE}/minizip/unzip.c ${XPWN_SOURCE}/minizip/ioapi.c)
target_include_directories(ltf_zip PUBLIC ${XPWN_SOURCE}/minizip)
target_link_libraries(ltf_zip PUBLIC ZLIB::ZLIB)
add_library(lighttouch_firmware SHARED src/firmware.cpp src/formats.cpp vendor/tinyxml2.cpp src/hfs-worker.c src/xpwn-guard.c
  ../../Packages/FirmwareKit/Sources/CActivation/activation.c)
target_compile_definitions(lighttouch_firmware PRIVATE LT_ACTIVATION_LIBRARY)
target_include_directories(lighttouch_firmware PUBLIC include PRIVATE vendor src
  ../../Packages/FirmwareKit/Sources/CActivation/include)
target_link_libraries(lighttouch_firmware PRIVATE ltf_dmg ltf_hfs ltf_zip OpenSSL::Crypto ZLIB::ZLIB)
if(ANDROID)
  target_link_options(lighttouch_firmware PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(NOT ANDROID)
  add_executable(lighttouch-prepare src/cli.cpp)
  target_link_libraries(lighttouch-prepare PRIVATE lighttouch_firmware)
  enable_testing()
  add_executable(firmware-formats-test tests/formats.cpp)
  target_include_directories(firmware-formats-test PRIVATE src vendor)
  target_compile_options(firmware-formats-test PRIVATE -UNDEBUG)
  target_link_libraries(firmware-formats-test PRIVATE lighttouch_firmware)
  target_compile_definitions(firmware-formats-test PRIVATE LT_METADATA_GOLDENS="${CMAKE_CURRENT_SOURCE_DIR}/tests/n72-metadata-goldens.txt")
  add_test(NAME firmware-formats COMMAND firmware-formats-test)
  add_executable(firmware-transaction-test tests/transaction.cpp)
  target_include_directories(firmware-transaction-test PRIVATE src vendor)
  target_compile_options(firmware-transaction-test PRIVATE -UNDEBUG)
  target_link_libraries(firmware-transaction-test PRIVATE lighttouch_firmware)
  add_test(NAME firmware-transaction COMMAND firmware-transaction-test)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(firmware-hfs-test tests/hfs-driver.cpp)
  target_include_directories(firmware-hfs-test PRIVATE src vendor)
  target_link_libraries(firmware-hfs-test PRIVATE lighttouch_firmware)
  add_test(NAME firmware-hfs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/hfs.py $<TARGET_FILE_DIR:firmware-hfs-test>)
endif()

foreach(target ltf_common ltf_hfs ltf_dmg)
  target_compile_definitions(${target} PRIVATE exit=lt_fw_xpwn_fatal)
  target_compile_options(${target} PRIVATE -Wno-deprecated-declarations)
endforeach()
